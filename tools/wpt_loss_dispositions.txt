# Accepted per-file pass losses, scoped to the generation that recorded them.
# Read by tools/wpt_progress.py. No entries are seeded: every observed loss is
# UNEXPLAINED until its evidence is reviewed and recorded here. A disposition
# explains a loss; it never changes passing counts or the blocking-file rule.
#
# Format: [worklist/path] followed by key = value lines. Blank lines and lines
# beginning with # are ignored. A path can occur in several generations; a
# (path, generation) pair can occur only once. Unknown, duplicate, empty or
# missing keys are errors. Every entry needs generation (positive integer),
# kind (one of the three below), and why (the explanation).
#
# Flake: three isolated BASE/TIP pairs, each reporting passes/total. State the
# commands, base/tip revisions and exact failing subtests in why (or link to
# the retained evidence there). The parser checks the three pair counts;
# review determines whether the evidence actually proves a flake.
#
#   [example/test.html]
#   generation = 170
#   kind = flake
#   why = explanation and the location of the isolated-run evidence
#   pair.1 = base 3/4; tip 4/4
#   pair.2 = base 4/4; tip 3/4
#   pair.3 = base 4/4; tip 4/4
#
# User ruling: a valid date and the user's decision, not an agent's inference.
#
#   [example/test.html]
#   generation = 171
#   kind = user-ruling
#   why = what changed and why the loss is accepted
#   date = 2026-10-20
#   decision = the user's decision and its source
#
# Upstream test change: two distinct full WPT commit IDs. They must match the
# loss's recorded before/after revisions to explain it. Unknown provenance is
# not evidence of an upstream change.
#
#   [example/test.html]
#   generation = 172
#   kind = upstream-test-change
#   why = what upstream removed or changed, with the upstream commit/PR
#   wpt.before = aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa
#   wpt.after = bbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbb
#
# Entries: none.
