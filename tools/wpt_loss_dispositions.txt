# Accepted per-file pass losses, scoped to the generation that recorded them.
# Read by tools/wpt_progress.py. No entries are seeded: every observed loss is
# UNEXPLAINED until its evidence is reviewed and recorded here. A disposition
# explains a loss; it never changes passing counts or the blocking-file rule.
#
# Format: [worklist/path] followed by key = value lines. Blank lines and lines
# beginning with # are ignored. A path can occur in several generations; a
# (path, generation) pair can occur only once. Unknown, duplicate, empty or
# missing keys are errors. Every entry needs generation (positive integer),
# kind (one of the five below), and why (the explanation).
#
# Flake: three isolated BASE/TIP pairs, each reporting passes/total. State the
# commands, base/tip revisions and exact failing subtests in why (or link to
# the retained evidence there). The parser checks the three pair counts;
# review determines whether the evidence actually proves a flake.
#
#   [example/test.html]
#   generation = 170
#   kind = flake
#   why = explanation and the location of the isolated-run evidence
#   pair.1 = base 3/4; tip 4/4
#   pair.2 = base 4/4; tip 3/4
#   pair.3 = base 4/4; tip 4/4
#
# User ruling: a valid date and the user's decision, not an agent's inference.
#
#   [example/test.html]
#   generation = 171
#   kind = user-ruling
#   why = what changed and why the loss is accepted
#   date = 2026-10-20
#   decision = the user's decision and its source
#
# Upstream test change: two distinct full WPT commit IDs. They must match the
# loss's recorded before/after revisions to explain it. Unknown provenance is
# not evidence of an upstream change.
#
#   [example/test.html]
#   generation = 172
#   kind = upstream-test-change
#   why = what upstream removed or changed, with the upstream commit/PR
#   wpt.before = aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa
#   wpt.after = bbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbb
#
# Fixed: the loss was a bug, and this is its fix. Under the one-flake rule a
# path that flakes again is a bug with an owner; once the owner fixes it, the
# fix is what explains the losses measured before it. commit is the fix (7 to
# 40 hex digits): the tool checks that it is a commit of this repository and
# an ancestor of HEAD (merged). test is the red test, one or more
# comma-separated repository paths that must be in the fix commit's own tree.
# repro is the failure count under the reproduction, before and after the fix:
# before must show the bug (at least one failure) and after must fail less
# often. The entry explains the loss ONLY for a generation whose head does not
# contain the fix commit: the same loss in a generation that has the fix is
# the bug again, unexplained (so is one whose head git cannot place). The
# loss history keeps the loss itself either way.
#
#   [example/test.html]
#   generation = 173
#   kind = fixed
#   why = the cause, the fix, and where the evidence is
#   commit = 0123456789ab
#   test = tests/example/red_test.zig
#   repro = before 12/200; after 0/200
#
# Timing-bound (the user, 2026-11-06): the file's result provably depends on
# whether a gap shorter than Crane's coarsening tick crosses a tick - two
# marks the spec takes back to back, floored to the 125 us grid
# (src/hr_time/coarsen.zig). tick_us is that tick (a positive integer);
# gap_us is the measured gap, "min <us> / p50 <us> / max <us>", whose median
# must be below tick_us; pass_rate is "<passes>/<runs>" under the condition,
# with both outcomes seen (0 < passes < runs); condition says what exposes
# it; measured_at is the commit measured (7 to 40 hex digits), which the
# tool checks is a commit of this repository and an ancestor of HEAD; paths
# lists the code the gap runs through, comma-separated, each in measured_at's
# tree. The entry explains the path's losses in its generation AND every
# later one (a later exact entry for the path wins), so a flip no longer
# blocks publication, and it puts the file on the progress page's
# timing-bound watch list with its evidence. A commit after measured_at that
# touches a listed path makes the entry STALE: a warning on the page and
# from the tool, never a block - re-measure the gap. It is not a flake: the
# one-flake rule does not count it, and it takes no isolated pairs.
#
#   [example/test.html]
#   generation = 205
#   kind = timing-bound
#   why = what the gap is, between which spec steps, and the evidence
#   tick_us = 125
#   gap_us = min 63 / p50 76 / max 100
#   pass_rate = 17/30
#   condition = what makes the gap decide the result
#   measured_at = 0123456789ab
#   paths = src/example/one.zig, src/example/two.zig
